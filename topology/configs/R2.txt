! R2 Base Configuration
! Role: Secondary router, BGP AS 65002
!
hostname R2
!
no ip domain-lookup
cdp run
!
! Loopback — router ID
interface Loopback0
 ip address 2.2.2.2 255.255.255.255
 no shutdown
!
! gi0/0 — BGP peering link to R1
interface GigabitEthernet0/0
 description BGP-PEERING-TO-R1
 ip address 10.255.0.2 255.255.255.252
 no shutdown
!
! gi0/1 — downlink to SW1
interface GigabitEthernet0/1
 description DOWNLINK-TO-SW1
 ip address 10.0.0.9 255.255.255.252
 no shutdown
!
! gi0/2 — downlink to SW2
interface GigabitEthernet0/2
 description DOWNLINK-TO-SW2
 ip address 10.0.0.13 255.255.255.252
 no shutdown
!
! Management interface
interface GigabitEthernet0/3
 description MANAGEMENT
 ip address 192.168.0.11 255.255.255.0
 no shutdown
!
! OSPF — internal routing
router ospf 1
 router-id 2.2.2.2
 network 2.2.2.2 0.0.0.0 area 0
 network 10.0.0.8 0.0.0.3 area 0
 network 10.0.0.12 0.0.0.3 area 0
!
! BGP — eBGP to R1
router bgp 65002
 bgp router-id 2.2.2.2
 bgp log-neighbor-changes
 neighbor 10.255.0.1 remote-as 65001
 neighbor 10.255.0.1 description eBGP-TO-R1
 !
 address-family ipv4
  neighbor 10.255.0.1 activate
  network 2.2.2.2 mask 255.255.255.255
  network 10.0.20.0 mask 255.255.255.0
 exit-address-family
!
username cisco privilege 15 secret cisco123
enable secret cisco
!
ip domain-name lab.local
crypto key generate rsa modulus 1024
ip ssh version 2
!
line vty 0 4
 login local
 transport input ssh
!
end
